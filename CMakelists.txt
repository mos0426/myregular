# 要求最低 CMake 版本
cmake_minimum_required(VERSION 4.3)

# 项目名称
project(MyRegular)

# 设置 C++ 标准
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 如果使用 C++20 的 std::format，可改为 20
# set(CMAKE_CXX_STANDARD 20)

# 设置可执行文件的输出目录（可选）
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)


#Lib utility
add_library(utility STATIC src/internal/utility/utf8.cpp)
target_include_directories(utility PUBLIC include/internal/utility)

# Lib expression
add_library(expression STATIC 
    src/internal/expression/error.cpp
    src/internal/expression/lexer.cpp
    src/internal/expression/parse.cpp
)
target_include_directories(expression PUBLIC include/internal/expression)
target_link_libraries(expression PRIVATE utility)

# Lib automata
add_library(automata STATIC 
    src/internal/automata/charset.cpp
    src/internal/automata/nfa.cpp
)
target_include_directories(automata PUBLIC include/internal/automata)

#Lib compile
add_library(compile STATIC src/internal/compile/compile_to_nfa.cpp)
target_include_directories(compile PUBLIC include/internal/compile)
target_link_libraries(compile PUBLIC automata expression)


# Unit test
add_executable(expression_test tests/internal/expression/parse_test.cpp)
target_link_libraries(expression_test PRIVATE expression GTest::gtest_main)

add_executable(utf8_test tests/internal/utility/utf8_test.cpp)
target_link_libraries(utf8_test PRIVATE utility GTest::gtest_main)

add_executable(automata_test 
    tests/internal/automata/nfa_test.cpp
    tests/internal/automata/charset_test.cpp
)
target_link_libraries(automata_test PRIVATE automata GTest::gtest_main)

add_executable(compile_test tests/internal/compile/compile_test.cpp)
target_link_libraries(compile_test PRIVATE compile GTest::gtest_main)


# 设置第三方库依赖
include(FetchContent)
FetchContent_Declare(
  googletest
  URL https://github.com/google/googletest/archive/03597a01ee50ed33e9dfd640b249b4be3799d395.zip
)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
enable_testing()


include(GoogleTest)
gtest_discover_tests(expression_test)
gtest_discover_tests(utf8_test)
gtest_discover_tests(automata_test)
gtest_discover_tests(compile_test)


# 如果使用 C++20 的 std::format，某些编译器需要链接额外的库（如 libstdc++fs 等），但通常不需要

target_compile_options(expression PRIVATE -Wall  -pedantic)

# MSVC 的警告选项
if(MSVC)
    target_compile_options(expression PRIVATE /W4)
    

endif()
